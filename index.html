Calendrier de réservation

#calendar { display: grid; grid-template-columns: repeat(7, 1fr); gap: 6px; max-width: 800px; } .day { padding: 12px; border: 1px solid #ccc; text-align: center; cursor: pointer; } .free { background: #e8ffe8; } .booked { background: #ffd6d6; } async function loadReservations() { const r = await fetch("api/load.php"); return await r.json(); } function renderCalendar(reservations) { const cal = document.getElementById("calendar"); cal.innerHTML = ""; const today = new Date(); const year = today.getFullYear(); const month = today.getMonth(); const firstDay = new Date(year, month, 1); const lastDay = new Date(year, month + 1, 0); for (let d = 1; d <= lastDay.getDate(); d++) { const dateStr = `${year}-${month+1}-${d}`; const div = document.createElement("div"); div.classList.add("day"); if (reservations.includes(dateStr)) { div.classList.add("booked"); } else { div.classList.add("free"); } div.innerHTML = d; div.onclick = async () => { await fetch("api/save.php?date=" + dateStr); location.reload(); }; cal.appendChild(div); } } loadReservations().then(renderCalendar);
